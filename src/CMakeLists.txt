cmake_minimum_required(VERSION 3.16)
project(YtGrab CXX RC)
set(CMAKE_CXX_STANDARD 17)
add_executable(YtGrab WIN32 main.cpp app.rc)
target_compile_definitions(YtGrab PRIVATE UNICODE _UNICODE)
if(MSVC)
  # Static CRT: no VC++ redistributable needed on the target machine.
  set_property(TARGET YtGrab PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  target_compile_options(YtGrab PRIVATE /utf-8 /O2 /W3)
  target_link_options(YtGrab PRIVATE /MANIFEST:NO)
else()
  target_link_options(YtGrab PRIVATE -municode -static -s)
endif()
target_link_libraries(YtGrab PRIVATE comctl32 uxtheme shell32 shlwapi ole32 uuid urlmon wininet dwmapi comdlg32 gdi32)
